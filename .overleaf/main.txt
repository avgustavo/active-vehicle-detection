                chunk_size=64

        t6 = time.time()
        print(f"Tempo total da geracao de predicoes no ciclo {i}: {calculate_time(t5, t6)}")
        ####################################################################################################
        printff(f"Tempo total de execucao do ciclo {i}: {calculate_time(t1, t6)}")
